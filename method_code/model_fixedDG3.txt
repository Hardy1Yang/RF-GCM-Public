    for (i in 1:n){
      for (k in 1:m){
        pX[i,k] <- ((z[k]*D)+((1-D)*g))
        X[i,k] ~ dbern(pX[i,k]) }}
      D ~ dbeta(Dmu*Dtau,(1-Dmu)*Dtau) T(0.001, 0.999)

    for (k in 1:m){
      z[k] ~ dbern(p) }

    Dmu ~ dbeta(alpha, alpha)
    alpha <- 2

                  gtaumu/pow(gtausig,2)) T(0.001, 999.999)
